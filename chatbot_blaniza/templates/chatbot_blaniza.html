Código de Entrenamiento del Modelo (QLoRA Fine-tuning)

import torch
import transformers
import pandas as pd
import numpy as np
import json
from collections import Counter
import matplotlib.pyplot as plt
import time
from datetime import datetime
import os

from transformers import AutoTokenizer, AutoModelForCausalLM, BitsAndBytesConfig, TrainingArguments, Trainer, DataCollatorForLanguageModeling
from peft import LoraConfig, get_peft_model, TaskType, prepare_model_for_kbit_training, PeftModel
from torch.utils.data import Dataset, DataLoader
import psutil

# Configuración del modelo y dataset
MODEL_NAME = "LenguajeNaturalAI/leniachat-qwen2-1.5B-v0"
DATASET_PATH = "dataset_mejorado_equilibrado_FIXED.jsonl"

# 1. Cargar y analizar el dataset
def load_and_analyze_dataset(dataset_path):
    data = []
    with open(dataset_path, 'r', encoding='utf-8') as f:
        for line in f:
            data.append(json.loads(line))
    
    system_prompts = [item['messages'][0]['content'] for item in data]
    user_questions = [item['messages'][1]['content'] for item in data]
    assistant_responses = [item['messages'][2]['content'] for item in data]
    
    tipos_qa = [item.get('qa_type', 'unknown') for item in data]
    contador_tipos = Counter(tipos_qa)
    
    return data, contador_tipos

# 2. Dataset personalizado para entrenamiento conversacional
class ConversationDataset(Dataset):
    def __init__(self, data, tokenizer, max_length=None):
        self.data = data
        self.tokenizer = tokenizer
        
        if max_length is None:
            self.max_length = self._calculate_optimal_max_length()
        else:
            self.max_length = max_length

    def _calculate_optimal_max_length(self):
        lengths = []
        for item in self.data:
            conversation = item['messages']
            system_msg = conversation[0]['content']
            user_msg = conversation[1]['content']
            assistant_msg = conversation[2]['content']
            
            text = f"system: {system_msg}\\nuser: {user_msg}\\nassistant: {assistant_msg}"
            tokens = self.tokenizer(text, return_tensors='pt')
            lengths.append(len(tokens['input_ids'][0]))
        
        lengths = np.array(lengths)
        optimal_length = np.max(lengths)
        return optimal_length

    def __len__(self):
        return len(self.data)

    def __getitem__(self, idx):
        conversation = self.data[idx]['messages']
        
        system_msg = conversation[0]['content']
        user_msg = conversation[1]['content']
        assistant_msg = conversation[2]['content']
        
        text = f"system: {system_msg}\\nuser: {user_msg}\\nassistant: {assistant_msg}"
        
        encoding = self.tokenizer(
            text,
            truncation=True,
            max_length=self.max_length,
            padding='max_length',
            return_tensors='pt'
        )
        
        return {
            'input_ids': encoding['input_ids'].flatten(),
            'attention_mask': encoding['attention_mask'].flatten(),
            'labels': encoding['input_ids'].flatten()
        }

# 3. Configurar modelo con cuantización 4-bit y LoRA
def setup_qlora_model(model_name):
    # Cargar tokenizer
    tokenizer = AutoTokenizer.from_pretrained(model_name)
    if tokenizer.pad_token is None:
        tokenizer.pad_token = tokenizer.eos_token
    
    # Configuración de cuantización 4-bit
    bnb_config = BitsAndBytesConfig(
        load_in_4bit=True,
        bnb_4bit_use_double_quant=True,
        bnb_4bit_quant_type="nf4",
        bnb_4bit_compute_dtype=torch.bfloat16
    )
    
    # Cargar modelo cuantizado
    model = AutoModelForCausalLM.from_pretrained(
        model_name,
        quantization_config=bnb_config,
        device_map="auto",
        trust_remote_code=True,
        use_cache=False
    )
    
    # Preparar modelo para k-bit training
    model = prepare_model_for_kbit_training(model)
    
    # Configuración LoRA optimizada para conversaciones
    lora_config = LoraConfig(
        r=64,                                 # Rank alto para mejor capacidad
        lora_alpha=32,                        # Alpha = r/2 (ratio recomendado)
        target_modules=["q_proj", "v_proj", "k_proj", "o_proj",
                       "gate_proj", "up_proj", "down_proj"],  # Todos los proyectores de atención
        lora_dropout=0.1,                     # Dropout moderado
        bias="none",                          # Sin bias adicional
        task_type=TaskType.CAUSAL_LM,         # Modelado de lenguaje causal
    )
    
    # Aplicar adaptadores LoRA
    model = get_peft_model(model, lora_config)
    
    return model, tokenizer

# 4. Configurar argumentos de entrenamiento QLoRA
def get_training_arguments():
    os.environ["WANDB_DISABLED"] = "true"
    
    training_args = TrainingArguments(
        output_dir="./blaniza-assistant-provisional",
        overwrite_output_dir=True,
        num_train_epochs=5,                   # Más épocas por ser más eficiente
        per_device_train_batch_size=8,        # Batch más alto gracias a QLoRA
        per_device_eval_batch_size=8,         # Batch más alto
        gradient_accumulation_steps=2,        # Batch efectivo = 16
        warmup_steps=50,                      # Warmup más largo
        weight_decay=0.01,
        logging_dir="./logs",
        logging_steps=5,
        eval_strategy="steps",
        eval_steps=15,                        # Evaluación más frecuente
        save_steps=30,                        # Guardado más frecuente
        save_total_limit=3,                   # Más checkpoints
        dataloader_pin_memory=False,
        remove_unused_columns=False,
        load_best_model_at_end=True,
        metric_for_best_model="eval_loss",
        greater_is_better=False,
        report_to=None,
        fp16=False,
        bf16=True,                            # BF16 compatible con QLoRA
        gradient_checkpointing=False,         # No necesario con QLoRA
        logging_nan_inf_filter=False,
        max_grad_norm=0.3,                    # Gradient clipping más agresivo para LoRA
        learning_rate=2e-4,                   # LR más alto típico para LoRA
        lr_scheduler_type="cosine",           # Cosine scheduling
        warmup_ratio=0.05,
        dataloader_num_workers=0,
        optim="paged_adamw_8bit",             # Optimizador 8-bit para QLoRA
    )
    
    return training_args

# 5. Entrenar modelo con QLoRA
def train_qlora_model(model, tokenizer, train_dataset, val_dataset, training_args):
    data_collator = DataCollatorForLanguageModeling(
        tokenizer=tokenizer,
        mlm=False,
        return_tensors="pt"
    )
    
    trainer = Trainer(
        model=model,
        args=training_args,
        train_dataset=train_dataset,
        eval_dataset=val_dataset,
        data_collator=data_collator,
        tokenizer=tokenizer,
    )
    
    if torch.cuda.is_available():
        torch.cuda.empty_cache()
    
    start_time = time.time()
    train_result = trainer.train()
    end_time = time.time()
    
    training_time = end_time - start_time
    
    # Guardar adaptadores LoRA
    model.save_pretrained("./blaniza-assistant")
    tokenizer.save_pretrained("./blaniza-assistant")
    
    return train_result, training_time

# 6. Fusionar adaptadores LoRA con modelo base para producción
def merge_lora_adapters(base_model_name, adapter_path, output_path):
    # Cargar modelo base sin cuantización
    base_model = AutoModelForCausalLM.from_pretrained(
        base_model_name,
        device_map="auto",
        trust_remote_code=True,
        torch_dtype=torch.bfloat16
    )
    
    # Cargar adaptadores LoRA
    model_with_lora = PeftModel.from_pretrained(base_model, adapter_path)
    
    # Fusionar adaptadores
    merged_model = model_with_lora.merge_and_unload()
    
    # Cargar tokenizer
    tokenizer = AutoTokenizer.from_pretrained(base_model_name)
    if tokenizer.pad_token is None:
        tokenizer.pad_token = tokenizer.eos_token
    
    # Guardar modelo fusionado
    os.makedirs(output_path, exist_ok=True)
    merged_model.save_pretrained(output_path, save_function=torch.save, max_shard_size="2GB")
    tokenizer.save_pretrained(output_path)
    
    return merged_model

# 7. Función principal de entrenamiento
def main():
    # Cargar dataset
    data, contador_tipos = load_and_analyze_dataset(DATASET_PATH)
    
    # Dividir dataset (80% entrenamiento, 20% validación)
    train_size = int(0.8 * len(data))
    train_data = data[:train_size]
    val_data = data[train_size:]
    
    # Configurar modelo QLoRA
    model, tokenizer = setup_qlora_model(MODEL_NAME)
    
    # Crear datasets
    train_dataset = ConversationDataset(train_data, tokenizer)
    val_dataset = ConversationDataset(val_data, tokenizer)
    
    # Configurar entrenamiento
    training_args = get_training_arguments()
    
    # Entrenar modelo
    train_result, training_time = train_qlora_model(
        model, tokenizer, train_dataset, val_dataset, training_args
    )
    
    # Fusionar adaptadores para producción
    merged_model = merge_lora_adapters(
        MODEL_NAME, 
        "./blaniza-assistant", 
        "./blaniza-assistant-merged"
    )
    
    return {
        "training_loss": train_result.training_loss,
        "training_time": training_time,
        "global_steps": train_result.global_step,
        "model_path": "./blaniza-assistant-merged"
    }

if __name__ == "__main__":
    results = main()
